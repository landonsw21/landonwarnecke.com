    const projects = DATA.projects
    const byId: Record<string, any> = {}
    projects.forEach((p: any, i: number) => { byId[p.id] = { ...p, next: projects[(i + 1) % projects.length] } })

    React.useEffect(() => {
        if (!pending.current) return
        const id = pending.current
        pending.current = null
        const el = document.getElementById(id)
        if (el) el.scrollIntoView({ behavior: "smooth", block: "start" })
    }, [open, tab])

    React.useEffect(() => {
        if (!open || typeof IntersectionObserver === "undefined") return
        const secs = Array.from(document.querySelectorAll('[data-chapter^="' + open + '-"]'))
        if (!secs.length) return
        const io = new IntersectionObserver((entries) => {
            const vis = entries.filter((e) => e.isIntersecting).sort((a, b) => a.boundingClientRect.top - b.boundingClientRect.top)
            if (vis.length) setActive(vis[0].target.getAttribute("data-chapter"))
        }, { rootMargin: "-15% 0px -65% 0px", threshold: 0 })
        secs.forEach((s) => io.observe(s))
        return () => io.disconnect()
    }, [open])

    const step = React.useCallback((d: number) => {
        setLb((cur) => (cur ? { list: cur.list, i: (cur.i + d + cur.list.length) % cur.list.length } : cur))
    }, [])
    React.useEffect(() => {
        if (!lb) return
        const onKey = (e: KeyboardEvent) => {
            if (e.key === "Escape") setLb(null)
            else if (e.key === "ArrowRight") step(1)
            else if (e.key === "ArrowLeft") step(-1)
        }
        window.addEventListener("keydown", onKey)
        return () => window.removeEventListener("keydown", onKey)
    }, [lb, step])

    const toggle = (id: string) => {
        const next = open === id ? null : id
        setOpen(next)
        setActive(next ? next + "-" + byId[next].sections[0].slug : null)
        pending.current = "p-" + id
    }
    const goNext = (id: string) => {
        const n = byId[id].next.id
        setOpen(n)
        setActive(n + "-" + byId[n].sections[0].slug)
        pending.current = "p-" + n
    }
    const go = (t: string) => {
        setTab(t); setOpen(null); setLb(null); setAbout(false)
        if (typeof window !== "undefined") window.scrollTo({ top: 0, behavior: "smooth" })
    }
    const jump = (e: React.MouseEvent, id: string) => {
        e.preventDefault()
        const el = document.getElementById(id)
        if (el) el.scrollIntoView({ behavior: "smooth", block: "start" })
    }
    const scrollRow = (sid: string, dir: number) => {
        const el = rows.current[sid]
        if (el) el.scrollBy({ left: dir * Math.max(320, el.clientWidth * 0.8), behavior: "smooth" })
    }
    const onRowScroll = (sid: string, e: React.UIEvent<HTMLDivElement>) => {
        const el = e.currentTarget
        const max = el.scrollWidth - el.clientWidth
        const p = max > 0 ? Math.round((el.scrollLeft / max) * 100) : 100
        setProg((cur) => (cur[sid] === p ? cur : { ...cur, [sid]: p }))
    }

    const Head = (h: { label: string; title: string; sub: string; right?: any }) => (
        <header className="lw-head">
            <div style={{ display: "flex", flexDirection: "column", gap: 10, maxWidth: 640 }}>
                <span className="lw-mono lw-label">{h.label}</span>
                <h1 className="lw-h1">{h.title}</h1>
                <p className="lw-sub">{h.sub}</p>
            </div>
            {h.right}
        </header>
    )

    const projectView = (
        <div>
            <Head label="SELECTED WORK · 2024–2026" title="Selected work"
                sub="Seven case studies and one project in progress. Open any project for the full story, from research to final." />
            <div className="lw-list">
                {projects.map((p: any, idx: number) => {
                    const isOpen = open === p.id
                    const sectionImgs = p.sections.flatMap((s: any) => s.imgs.map((im: any) => ({ src: img(im[0]), alt: im[1], cap: im[1] })))
                    return (
                        <article key={p.id} id={"p-" + p.id} style={{ scrollMarginTop: 16 }}>
                            <button className="lw-card" onClick={() => toggle(p.id)} aria-expanded={isOpen} aria-controls={"d-" + p.id}>
                                {p.soon ? (
                                    <span className="lw-soon"><img src={img(p.hero)} alt={p.alt} loading="lazy" /><span className="lw-mono">COMING SOON</span></span>
                                ) : (
                                    <img className="lw-card-img" src={img(p.hero)} alt={p.alt} loading={idx < 2 ? "eager" : "lazy"} decoding="async" />
                                )}
                                <span className="lw-card-body">
                                    <span className="lw-card-text">
                                        <span className="lw-mono" style={{ fontSize: 12 }}>{p.num} · {p.kind}</span>
                                        <span className="lw-card-name">{p.name}</span>
                                        <span className="lw-card-line">{p.line}</span>
                                    </span>
                                    <span className="lw-pill">{isOpen ? "Close −" : p.soon ? "Preview +" : "Open case study +"}</span>
                                </span>
                            </button>
                            {isOpen && (
                                <div id={"d-" + p.id} className="lw-drawer">
                                    <nav className="lw-chapbar" aria-label={p.name + " sections"}>
                                        <div className="lw-chaps">
                                            {p.sections.map((s: any) => {
                                                const cid = p.id + "-" + s.slug
                                                return <a key={cid} href={"#" + cid} onClick={(e) => jump(e, cid)} className={"lw-chap" + (active === cid ? " is-active" : "")}>{s.title}</a>
                                            })}
                                        </div>
                                        <button className="lw-close" onClick={() => toggle(p.id)} aria-label={"Close " + p.name}>Close ×</button>
                                    </nav>
                                    <div className="lw-dbody">
                                        <dl className="lw-meta">
                                            {p.meta.map((m: any) => (
                                                <div key={m[0]}><dt className="lw-mono">{m[0].toUpperCase()}</dt><dd>{m[1]}</dd></div>
                                            ))}
                                        </dl>
                                        {p.sections.map((s: any, i: number) => {
                                            const cid = p.id + "-" + s.slug
                                            return (
                                                <section key={cid} id={cid} data-chapter={cid} className="lw-sec">
                                                    <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
                                                        <span className="lw-mono lw-label">{pad(i + 1)}</span>
                                                        <h3>{s.title}</h3>
                                                    </div>
                                                    <div className="lw-sec-body">
                                                        {s.paras.map((t: string, j: number) => <p key={j} dangerouslySetInnerHTML={{ __html: t }} />)}
                                                        {s.imgs.length > 0 && (
                                                            <div className={"lw-imgs" + (s.cols === 2 && s.imgs.length > 1 ? " c2" : "")}>
                                                                {s.imgs.map((im: any) => (
                                                                    <img key={im[0]} src={img(im[0])} alt={im[1]} loading="lazy" decoding="async"
                                                                        onClick={() => setLb({ list: sectionImgs, i: sectionImgs.findIndex((x: any) => x.src === img(im[0])) })} />
                                                                ))}
                                                            </div>
                                                        )}
                                                    </div>
                                                </section>
                                            )
                                        })}
                                        <div className="lw-dfoot">
                                            <button className="lw-ghost" onClick={() => toggle(p.id)}>Close project ↑</button>
                                            <button className="lw-pill" onClick={() => goNext(p.id)}>Next: {byId[p.id].next.name} →</button>
                                        </div>
                                    </div>
                                </div>
                            )}
                        </article>
                    )
                })}
            </div>
        </div>
    )

    const extrasView = (
        <div>
            <Head label="EXTRAS" title="Tools I’ve built" sub="Small tools I made for my own process, free to try in your browser." />
            <div className="lw-two">
                {DATA.extras.map((x: any) => (
                    <article key={x.name} className="lw-extra">
                        <img src={img(x.img)} alt={x.alt} loading="lazy" />
                        <div className="lw-extra-body">
                            <span className="lw-mono" style={{ fontSize: 12 }}>{x.tag}</span>
                            <h2>{x.name}</h2>
                            <p>{x.desc}</p>
                            <a className="lw-pill" style={{ marginTop: "auto", alignSelf: "flex-start", color: "#FFFFFF" }} href={x.url} target="_blank" rel="noopener">Open tool ↗</a>
                        </div>
                    </article>
                ))}
            </div>
        </div>
    )

    const galleryView = (
        <div>
            <Head label="GALLERY" title="Gallery" sub="Drawings, making and smaller projects. Drag, swipe or use the arrows; tap any piece to view it large." />
            <div style={{ display: "flex", flexDirection: "column", gap: 48 }}>
                {DATA.gallery.map((g: any, gi: number) => {
                    const list = g.items.map((it: any) => ({ src: img(it[0]), alt: it[1], cap: it[1] }))
                    return (
                        <section key={g.id} className="lw-gsec" aria-labelledby={"g-" + g.id}>
                            <div className="lw-ghead">
                                <div style={{ display: "flex", flexDirection: "column", gap: 6 }}>
                                    <span className="lw-mono lw-label">{pad(gi + 1)} · {g.items.length} PIECES</span>
                                    <h2 id={"g-" + g.id}>{g.title}</h2>
                                </div>
                                <div style={{ display: "flex", gap: 8 }}>
                                    <button className="lw-icon" onClick={() => scrollRow(g.id, -1)} aria-label={"Scroll " + g.title + " left"}>←</button>
                                    <button className="lw-icon" onClick={() => scrollRow(g.id, 1)} aria-label={"Scroll " + g.title + " right"}>→</button>
                                </div>
                            </div>
                            <div className="lw-row" ref={(el) => { rows.current[g.id] = el }} onScroll={(e) => onRowScroll(g.id, e)}>
                                {list.map((it: any, i: number) => (
                                    <button key={it.src} className="lw-gcard" onClick={() => setLb({ list, i })} aria-label={"View " + it.cap}>
                                        <img src={it.src} alt="" loading="lazy" decoding="async" />
                                        <span className="lw-gcap"><span className="lw-mono lw-label" style={{ paddingTop: 2 }}>{pad(i + 1)}</span><span>{it.cap}</span></span>
                                    </button>
                                ))}
                            </div>
                            <div className="lw-track"><div style={{ width: Math.max(6, prog[g.id] || 0) + "%" }} /></div>
                        </section>
                    )
                })}
            </div>
        </div>
    )

    const R = DATA.resume
    const rrow = (e: any) => (
        <div key={e[1]} className="lw-rrow">
            <span className="lw-mono lw-label" style={{ paddingTop: 3 }}>{e[0]}</span>
            <div style={{ display: "flex", flexDirection: "column", gap: 3 }}><b>{e[1]}</b><small>{e[2]}</small></div>
        </div>
    )
    const resumeView = (
        <div id="resume">
            <Head label="RESUME" title="Resume" sub="Industrial design student at Western Washington University. [TO CONFIRM: graduation year]"
                right={
                    <div style={{ display: "flex", flexDirection: "column", alignItems: "flex-end", gap: 6 }}>
                        <a className="lw-pill" style={{ color: "#FFFFFF" }} href="#resume">Download PDF ↓</a>
                        <span className="lw-mono lw-label" style={{ fontSize: 11 }}>[PDF LINK TO ADD]</span>
                    </div>
                } />
            <div className="lw-two" style={{ gap: 40 }}>
                <div style={{ display: "flex", flexDirection: "column", gap: 36 }}>
                    <section className="lw-rblock"><h2>Education</h2>{R.education.map(rrow)}</section>
                    <section className="lw-rblock"><h2>Experience</h2>{R.experience.map(rrow)}</section>
                    <section className="lw-rblock"><h2>Activities</h2>{R.activities.map(rrow)}</section>
                </div>
                <div style={{ display: "flex", flexDirection: "column", gap: 36 }}>
                    <section className="lw-rblock"><h2>Skills</h2>
                        {R.skills.map((s: any) => (
                            <div key={s[0]} className="lw-rrow" style={{ gridTemplateColumns: "170px minmax(0,1fr)" }}>
                                <b style={{ fontSize: 14 }}>{s[0]}</b><span style={{ fontSize: 15, lineHeight: 1.5 }}>{s[1]}</span>
                            </div>
                        ))}
                    </section>
                    <section className="lw-rblock"><h2>Languages</h2><p className="lw-rrow" style={{ display: "block", margin: 0, fontSize: 15 }}>{R.languages}</p></section>
                    <section className="lw-rblock"><h2>Interests</h2><p className="lw-rrow" style={{ display: "block", margin: 0, fontSize: 15, lineHeight: 1.55 }}>{R.interests}</p></section>
                </div>
            </div>
        </div>
    )

    return (
        <div className="lw-root" style={{ fontFamily: font }}>
            <style>{CSS}</style>
            <div className="lw-shell">
                <aside className={"lw-side" + (about ? " is-about" : "")} aria-label="About and navigation">
                    <div className="lw-id">
                        <img className="lw-portrait" src={img("portrait")} alt="Portrait of Landon Schmidt-Warnecke" />
                        <div style={{ display: "flex", flexDirection: "column", gap: 4, minWidth: 0 }}>
                            <span className="lw-nm">Landon Schmidt-Warnecke</span>
                            <span className="lw-role">Industrial Design · Western Washington University</span>
                        </div>
                        <button className="lw-about-toggle" onClick={() => setAbout(!about)} aria-expanded={about}>{about ? "Hide about" : "About me"}</button>
                    </div>
                    <div className="lw-bio"><span className="lw-mono lw-label">ABOUT ME</span><p>{DATA.bio}</p></div>
                    <nav className="lw-tabs" aria-label="Sections">
                        {TABS.map(([k, label], i) => (
                            <button key={k} className={"lw-tab" + (tab === k ? " is-active" : "")} onClick={() => go(k)} aria-current={tab === k ? "page" : undefined}>
                                <span>{label}</span><span className="lw-mono lw-tabnum">{pad(i + 1)}</span>
                            </button>
                        ))}
                    </nav>
                    <div className="lw-side-foot">
                        <span className="lw-mono lw-label">CONTACT</span>
                        <span>[TO ADD: email and LinkedIn]</span>
                        <span style={{ fontSize: 13, color: "#6A5F58" }}>© 2026 Landon Schmidt-Warnecke</span>
                    </div>
                </aside>
                <main className="lw-main">
                    {tab === "projects" && projectView}
                    {tab === "extras" && extrasView}
                    {tab === "gallery" && galleryView}
                    {tab === "resume" && resumeView}
                </main>
            </div>
            {lb && (
                <section className="lw-lb" aria-label="Image viewer">
                    <button className="lw-lb-bg" onClick={() => setLb(null)} aria-label="Close image viewer" />
                    <figure>
                        <img src={lb.list[lb.i].src} alt={lb.list[lb.i].alt} />
                        <figcaption><span className="lw-mono" style={{ fontSize: 12, color: "rgba(255,255,255,.75)" }}>{lb.i + 1} / {lb.list.length}</span><span>{lb.list[lb.i].cap}</span></figcaption>
                    </figure>
                    <button className="lw-lbicon" style={{ left: 18, top: "50%", marginTop: -24 }} onClick={() => step(-1)} aria-label="Previous image">←</button>
                    <button className="lw-lbicon" style={{ right: 18, top: "50%", marginTop: -24 }} onClick={() => step(1)} aria-label="Next image">→</button>
                    <button className="lw-lbicon" style={{ right: 18, top: 18 }} onClick={() => setLb(null)} aria-label="Close">×</button>
                </section>
            )}
        </div>
    )
}

addPropertyControls(Portfolio, {
    font: { type: ControlType.Enum, title: "Font", options: ["Manrope", "Archivo", "Space Grotesk"], defaultValue: "Manrope" },
    imageBase: { type: ControlType.String, title: "Image base URL", defaultValue: DEFAULT_BASE },
})
